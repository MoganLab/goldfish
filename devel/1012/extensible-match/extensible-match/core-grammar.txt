core-pattern ::= (core:wildcard)    ;; match anything, don’t bind anything
              |  (core:var id)      ;; match anything, bind variable
              |  (core:quote datum) ;; match something equal? to datum
              |  (core:and core-pattern_1 core-pattern_2)
                                    ;; match core-pattern_1 first, and
                                    ;; if it matches, also match
                                    ;; core-pattern_2
              |  (core:or core-pattern_1 core-pattern_2)
                                    ;; match core-pattern_1, succeed
                                    ;; if it matches; otherwise try
                                    ;; core-pattern_2
              |  (core:row core-pattern ...)
                                    ;; match all of the core-patterns,
                                    ;; but in any order
              |  (core:subject id core-pattern)
                                    ;; use the variable id as the
                                    ;; subject of the core-pattern
              |  (core:not core-pattern)
                                    ;; match something other than the
                                    ;; pattern; no variables bound
              |  (core:? pred)      ;; match when calling proc
                                    ;; on value returns true
              |  (core:apply proc (var ...) subpat)
                                    ;; match when calling proc on subject
                                    ;; returns as many vals as there are
                                    ;; vars, and the subpat (using
                                    ;; core:subject to extract the
                                    ;; respective vals from the vars)
                                    ;; matches
              |  (core:seq seq-type name ((var init step) ...)
                           terminate? ref seq-pattern ...)
                                    ;; match a sequence


seq-type ::= core/seq:ordered
          |  core/seq:partial
          |  core/seq:unordered

seq-pattern ::= (core/seq:one core-pattern)
                                    ;; match one item which matches
                                    ;; core-pattern
             |  (core/seq:many min max core-pattern)
                                    ;; match between min and max items
                                    ;; all of which match core-pattern;
                                    ;; variables within core-pattern are
                                    ;; bound as lists

min ::= (an exact, nonnegative integer)
max ::= (an exact, nonnegative integer not less than min)
     |  #t    ;; match at least min items, no upper limit
